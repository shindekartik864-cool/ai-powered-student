import React, { useState, useRef, useEffect } from 'react';
import { 
  Bot, 
  Send, 
  Sparkles, 
  Volume2, 
  VolumeX, 
  User, 
  Search, 
  AlertCircle, 
  RefreshCw, 
  CheckCircle2, 
  HelpCircle,
  Stethoscope,
  MessageSquare
} from 'lucide-react';
import { Language, translations } from '../translations';
import { MAHARASHTRA_CROPS } from '../data/maharashtraData';
import { speakText, stopSpeaking } from '../utils/speech';

interface Message {
  id: string;
  role: 'user' | 'model';
  text: string;
  timestamp: string;
}

interface AiChatbotProps {
  language: Language;
  farmData: any;
  onTriggerSpeech: (text: string) => void;
}

export const AiChatbot: React.FC<AiChatbotProps> = ({
  language,
  farmData,
  onTriggerSpeech,
}) => {
  const t = translations[language];

  const [activeTab, setActiveTab] = useState<'chat' | 'diagnose'>('chat');
  const [messages, setMessages] = useState<Message[]>([
    {
      id: 'welcome',
      role: 'model',
      text: language === 'mr'
        ? `🌱 **नमस्कार शेतकरी बंधू!** मी तुमचा डिजिटल कृषी मित्र **ग्रामसाथी AI** आहे.
कापूस, सोयाबीन, कांदा, ऊस, डाळिंब या पिकांवरील कीड-रोग नियंत्रण, खत व्यवस्थापन, फवारणी वेळापत्रक किंवा बाजारभावाबाबत कोणताही प्रश्न विचारा. मी तुम्हाला कृषी विद्यापीठांच्या मानकांनुसार योग्य सल्ला देईन!`
        : language === 'hi'
        ? `🌱 **नमस्कार किसान भाई!** मैं आपका डिजिटल कृषि सहायक **ग्रामसाथी AI** हूँ।
कपास, सोयाबीन, प्याज, गन्ना या अन्य फसलों के रोग नियंत्रण, खाद शेड्यूल या मंडी भाव संबंधी कोई भी सवाल पूछें!`
        : `🌱 **Welcome Farmer!** I am your **GramSathi AI** digital agronomy assistant calibrated for Maharashtra agriculture.
Ask me any questions on pest control, fertilizer dosages, crop scheduling, or APMC mandi trends!`,
      timestamp: 'आत्ताच',
    },
  ]);

  const [inputText, setInputText] = useState('');
  const [loading, setLoading] = useState(false);
  const [currentlySpeakingId, setCurrentlySpeakingId] = useState<string | null>(null);

  // Diagnostic tab state
  const [diagCrop, setDiagCrop] = useState(farmData.activeCropId || 'onion');
  const [diagSymptoms, setDiagSymptoms] = useState('');
  const [diagResult, setDiagResult] = useState<string | null>(null);
  const [loadingDiag, setLoadingDiag] = useState(false);

  const messagesEndRef = useRef<HTMLDivElement>(null);

  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  useEffect(() => {
    scrollToBottom();
  }, [messages, loading]);

  const handleSendMessage = async (textToSend?: string) => {
    const text = (textToSend || inputText).trim();
    if (!text || loading) return;

    const userMsg: Message = {
      id: Date.now().toString(),
      role: 'user',
      text,
      timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
    };

    setMessages((prev) => [...prev, userMsg]);
    setInputText('');
    setLoading(true);

    try {
      const history = messages.slice(-6).map((m) => ({
        role: m.role,
        text: m.text,
      }));

      const res = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          message: text,
          history,
          language,
          farmContext: {
            farmer: farmData.farmerName,
            district: farmData.district,
            acres: farmData.totalAcres,
            crops: `${farmData.activeCropId}, ${farmData.secondaryCropId}`,
            soil: farmData.soilType,
            irrigation: farmData.irrigationMethod,
          },
        }),
      });

      const data = await res.json();
      const botMsg: Message = {
        id: (Date.now() + 1).toString(),
        role: 'model',
        text: data.reply || 'माफ करा, उत्तर मिळवण्यात अडचण आली.',
        timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
      };

      setMessages((prev) => [...prev, botMsg]);
    } catch (err) {
      console.error('Chat error:', err);
      setMessages((prev) => [
        ...prev,
        {
          id: (Date.now() + 1).toString(),
          role: 'model',
          text: '🌾 **कृषी सल्ला:** सद्यस्थितीत तांत्रिक कारणामुळे उत्तर देता आले नाही. कृपया पुन्हा प्रयत्न करा किंवा स्थानिक KVK केंद्राशी संपर्क साधा.',
          timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
        },
      ]);
    } finally {
      setLoading(false);
    }
  };

  const handleDiagnose = async () => {
    if (!diagSymptoms.trim() || loadingDiag) return;
    setLoadingDiag(true);
    try {
      const cropObj = MAHARASHTRA_CROPS.find((c) => c.id === diagCrop);
      const res = await fetch('/api/diagnose', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          crop: cropObj?.nameMr || diagCrop,
          symptoms: diagSymptoms,
          language,
        }),
      });
      const data = await res.json();
      setDiagResult(data.diagnosis || 'निदान उपलब्ध नाही.');
    } catch (err) {
      console.error('Diag error:', err);
      setDiagResult('निदान प्रक्रियेत त्रुटी आली. कृपया स्थानिक कृषी सहाय्यक किंवा KVK कडून तपासणी करून घ्यावी.');
    } finally {
      setLoadingDiag(false);
    }
  };

  const handleSpeak = (id: string, text: string) => {
    if (currentlySpeakingId === id) {
      stopSpeaking();
      setCurrentlySpeakingId(null);
    } else {
      stopSpeaking();
      setCurrentlySpeakingId(id);
      onTriggerSpeech(text);
      speakText(text, language, () => setCurrentlySpeakingId(null));
    }
  };

  const commonSymptomChips = [
    'पानांवर जांभळे लांबट चट्टे पडत आहेत (Purple Blotch)',
    'पाने पिवळी पडून कडेने वाकडी होत आहेत (Thrips/Sucking pests)',
    'खोडावर छिद्र पडून विष्ठा दिसत आहे (Stem borer)',
    'झाड अचानक कोमेजत आहे व मुळे सडत आहेत (Root rot / Wilt)',
    'शेंगांना किंवा बोंडांना छिद्रे दिसत आहेत (Pod borer / Bollworm)',
  ];

  return (
    <div className="space-y-6 pb-12">
      {/* Title Header with Mode Switcher */}
      <div className="rounded-2xl bg-white border border-slate-200/80 p-5 sm:p-6 shadow-xs flex flex-col sm:flex-row sm:items-center justify-between gap-4">
        <div>
          <div className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full bg-emerald-100 text-emerald-800 text-xs font-bold mb-2">
            <Bot className="w-3.5 h-3.5 text-emerald-600" />
            <span>{t.chatTitle}</span>
          </div>
          <h2 className="text-xl sm:text-2xl font-extrabold text-slate-900 tracking-tight">
            {t.chatTitle}
          </h2>
          <p className="text-xs sm:text-sm text-slate-500 mt-1">
            {t.chatSubtitle}
          </p>
        </div>

        <div className="flex items-center gap-1 bg-slate-100 p-1 rounded-xl text-xs font-bold">
          <button
            onClick={() => setActiveTab('chat')}
            className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg transition ${
              activeTab === 'chat' ? 'bg-white text-emerald-800 shadow-xs' : 'text-slate-600 hover:text-slate-900'
            }`}
          >
            <MessageSquare className="w-3.5 h-3.5" />
            <span>कृषी संवाद (Chat)</span>
          </button>
          <button
            onClick={() => setActiveTab('diagnose')}
            className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg transition ${
              activeTab === 'diagnose' ? 'bg-white text-emerald-800 shadow-xs' : 'text-slate-600 hover:text-slate-900'
            }`}
          >
            <Stethoscope className="w-3.5 h-3.5 text-emerald-600" />
            <span>{t.diagnoseTab}</span>
          </button>
        </div>
      </div>

      {activeTab === 'chat' ? (
        <div className="rounded-2xl bg-white border border-slate-200/80 shadow-xs overflow-hidden flex flex-col h-[650px]">
          {/* Chat Messages Scrollable Box */}
          <div className="flex-1 overflow-y-auto p-4 sm:p-6 space-y-4 bg-slate-50/50">
            {messages.map((m) => {
              const isUser = m.role === 'user';
              return (
                <div
                  key={m.id}
                  className={`flex items-start gap-2.5 ${isUser ? 'flex-row-reverse' : 'flex-row'}`}
                >
                  {/* Avatar */}
                  <div className={`w-8 h-8 rounded-full flex items-center justify-center shrink-0 text-white font-bold text-xs ${
                    isUser ? 'bg-slate-700' : 'bg-emerald-600 shadow-xs'
                  }`}>
                    {isUser ? <User className="w-4 h-4" /> : <Bot className="w-4 h-4" />}
                  </div>

                  {/* Message Bubble */}
                  <div className={`max-w-[85%] sm:max-w-[75%] rounded-2xl p-4 text-xs sm:text-sm leading-relaxed ${
                    isUser 
                      ? 'bg-slate-900 text-white rounded-tr-xs shadow-xs' 
                      : 'bg-white border border-emerald-100 text-slate-800 rounded-tl-xs shadow-xs'
                  }`}>
                    <div className="whitespace-pre-line font-medium">
                      {m.text}
                    </div>

                    <div className="mt-2.5 pt-2 border-t border-slate-100 flex items-center justify-between text-[10px] text-slate-400">
                      <span>{m.timestamp}</span>
                      {!isUser && (
                        <div className="flex items-center gap-2">
                          <span className="font-semibold text-emerald-700 bg-emerald-50 px-1.5 py-0.2 rounded-sm">
                            {t.aiBadge}
                          </span>
                          <button
                            onClick={() => handleSpeak(m.id, m.text)}
                            className="text-slate-500 hover:text-emerald-700 p-1 rounded-md transition"
                            title="Read message aloud"
                          >
                            {currentlySpeakingId === m.id ? (
                              <VolumeX className="w-3.5 h-3.5 text-amber-600" />
                            ) : (
                              <Volume2 className="w-3.5 h-3.5" />
                            )}
                          </button>
                        </div>
                      )}
                    </div>
                  </div>
                </div>
              );
            })}

            {loading && (
              <div className="flex items-center gap-2 text-xs font-semibold text-emerald-700 bg-emerald-50 p-3 rounded-xl max-w-xs animate-pulse">
                <Sparkles className="w-4 h-4 text-emerald-600 animate-spin" />
                <span>ग्रामसाथी AI उत्तर तयार करत आहे...</span>
              </div>
            )}
            <div ref={messagesEndRef} />
          </div>

          {/* Quick Questions Chips */}
          <div className="p-3 bg-white border-t border-slate-100 overflow-x-auto flex items-center gap-2 text-xs">
            <span className="text-[11px] font-bold text-slate-400 whitespace-nowrap pl-1">
              वारंवार प्रश्न:
            </span>
            {[t.prompt1, t.prompt2, t.prompt3, t.prompt4].map((prompt, idx) => (
              <button
                key={idx}
                onClick={() => handleSendMessage(prompt)}
                className="px-2.5 py-1 rounded-lg bg-slate-100 hover:bg-emerald-50 hover:text-emerald-800 text-slate-700 whitespace-nowrap transition text-xs font-medium border border-slate-200/60"
              >
                {prompt}
              </button>
            ))}
          </div>

          {/* Message Input Box */}
          <div className="p-3 sm:p-4 bg-white border-t border-slate-200">
            <form
              onSubmit={(e) => {
                e.preventDefault();
                handleSendMessage();
              }}
              className="flex items-center gap-2"
            >
              <input
                type="text"
                value={inputText}
                onChange={(e) => setInputText(e.target.value)}
                placeholder={t.inputPlaceholder}
                className="flex-1 px-4 py-2.5 rounded-xl border border-slate-200 text-xs sm:text-sm text-slate-800 placeholder-slate-400 focus:outline-none focus:ring-2 focus:ring-emerald-500 font-medium"
              />
              <button
                type="submit"
                disabled={loading || !inputText.trim()}
                className="px-4 py-2.5 rounded-xl bg-emerald-600 hover:bg-emerald-700 disabled:opacity-50 text-white font-bold text-xs sm:text-sm shadow-xs transition flex items-center gap-1.5"
              >
                <span>{t.send}</span>
                <Send className="w-3.5 h-3.5" />
              </button>
            </form>
          </div>
        </div>
      ) : (
        /* Disease & Symptoms Diagnostic Simulator Tab */
        <div className="rounded-2xl bg-white border border-slate-200/80 p-5 sm:p-6 shadow-xs space-y-6">
          <div className="space-y-1">
            <h3 className="font-extrabold text-slate-900 text-base flex items-center gap-2">
              <Stethoscope className="w-5 h-5 text-emerald-600" />
              <span>{t.diagnoseTab} (Crop Disease Diagnostic)</span>
            </h3>
            <p className="text-xs text-slate-500">
              पिकाचे नाव व दिसणारी लक्षणे निवडा किंवा लिहा; AI संभाव्य कीड/रोग आणि जैविक व रासायनिक उपाय सांगेल.
            </p>
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            <div>
              <label className="block text-xs font-bold text-slate-700 mb-1">
                {t.selectCrop}
              </label>
              <select
                value={diagCrop}
                onChange={(e) => setDiagCrop(e.target.value)}
                className="w-full px-3 py-2 rounded-xl border border-slate-200 text-xs sm:text-sm font-bold text-slate-800 focus:outline-none focus:ring-2 focus:ring-emerald-500"
              >
                {MAHARASHTRA_CROPS.map((c) => (
                  <option key={c.id} value={c.id}>{c.nameMr} ({c.nameEn})</option>
                ))}
              </select>
            </div>

            <div>
              <label className="block text-xs font-bold text-slate-700 mb-1">
                पिकाचा भाग (Affected Plant Part)
              </label>
              <select className="w-full px-3 py-2 rounded-xl border border-slate-200 text-xs sm:text-sm font-medium text-slate-800 focus:outline-none focus:ring-2 focus:ring-emerald-500">
                <option>पाने (Leaves)</option>
                <option>खोड (Stem)</option>
                <option>फळे किंवा शेंगा (Fruits / Pods)</option>
                <option>मुळे (Roots)</option>
                <option>संपूर्ण झाड (Whole Plant)</option>
              </select>
            </div>
          </div>

          <div>
            <label className="block text-xs font-bold text-slate-700 mb-1.5">
              {t.enterSymptoms}
            </label>
            <div className="flex flex-wrap gap-2 mb-2">
              {commonSymptomChips.map((chip, idx) => (
                <button
                  key={idx}
                  type="button"
                  onClick={() => setDiagSymptoms(chip)}
                  className="px-2.5 py-1 rounded-lg bg-slate-100 hover:bg-emerald-50 text-slate-700 text-[11px] font-medium border border-slate-200 transition"
                >
                  + {chip}
                </button>
              ))}
            </div>

            <textarea
              rows={3}
              value={diagSymptoms}
              onChange={(e) => setDiagSymptoms(e.target.value)}
              placeholder="उदा. कांद्याची पात पिवळी पडून टोके वाकत आहेत व त्यावर काळे-जांभळे ठिपके दिसत आहेत..."
              className="w-full p-3 rounded-xl border border-slate-200 text-xs sm:text-sm text-slate-800 placeholder-slate-400 focus:outline-none focus:ring-2 focus:ring-emerald-500 font-medium"
            />
          </div>

          <button
            onClick={handleDiagnose}
            disabled={loadingDiag || !diagSymptoms.trim()}
            className="w-full py-3 px-4 rounded-xl bg-gradient-to-r from-emerald-600 to-green-600 hover:from-emerald-700 hover:to-green-700 text-white font-bold text-xs sm:text-sm shadow-md transition flex items-center justify-center gap-2 active:scale-98 disabled:opacity-50"
          >
            {loadingDiag ? (
              <>
                <Sparkles className="w-4 h-4 animate-spin text-amber-300" />
                <span>रोग तपासणी चालू आहे...</span>
              </>
            ) : (
              <>
                <Stethoscope className="w-4 h-4" />
                <span>{t.diagnoseBtn}</span>
              </>
            )}
          </button>

          {diagResult && (
            <div className="p-4 sm:p-5 rounded-xl bg-emerald-50/70 border border-emerald-200 space-y-3">
              <div className="flex items-center justify-between">
                <span className="text-xs font-extrabold uppercase text-emerald-900">
                  AI रोग निदान व कृषी उपाय
                </span>
                <button
                  onClick={() => handleSpeak('diag', diagResult)}
                  className="p-1 rounded-md bg-white border border-emerald-300 text-emerald-700 hover:bg-emerald-50 transition"
                  title="Listen to diagnosis"
                >
                  <Volume2 className="w-4 h-4" />
                </button>
              </div>

              <div className="text-xs sm:text-sm text-slate-800 whitespace-pre-line leading-relaxed font-medium">
                {diagResult}
              </div>

              <div className="pt-2 border-t border-emerald-200/60 text-[10px] text-slate-500 italic">
                {t.aiAdvisoryNote}
              </div>
            </div>
          )}
        </div>
      )}
    </div>
  );
};
